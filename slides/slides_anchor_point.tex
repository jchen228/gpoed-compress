% ── SLIDE 2: The Idea ────────────────────────────────────────
\begin{frame}{The Idea}
\vspace{0.3em}
\begin{columns}[T, totalwidth=\textwidth]

  \begin{column}{0.32\textwidth}
    \begin{block}{\stepcircle{A}\hspace{4pt} Place anchor points}
      Each round, select $k{=}200$ anchor points from the remaining
      domain via randomised pivoted Cholesky — targeting
      high-uncertainty regions.
    \end{block}
  \end{column}

  \begin{column}{0.32\textwidth}
    \begin{block}{\stepcircle{GP}\hspace{4pt} GP prediction}
      Fit a Gaussian Process on accumulated anchor values. Predict
      the full field from 200, 400, 600\ldots{} anchors as rounds
      progress.
    \end{block}
  \end{column}

  \begin{column}{0.32\textwidth}
    \begin{block}{\stepcircle{$\checkmark$}\hspace{4pt} Accept \& compress}
      Points predicted within $\pm\varepsilon_b$ of truth are
      \textbf{compressed} and removed from the pool. The rest remain
      for the next round.
    \end{block}
  \end{column}

\end{columns}
\vfill
{\scriptsize\color{cmuted}\itshape
  Rounds repeat until all points are compressed or the budget is exhausted.
  Anchor values stored losslessly; all others reconstructed via GP at decode time.}
\end{frame}

% ── SLIDE 3: Algorithm ───────────────────────────────────────
\begin{frame}{Algorithm}
\vspace{0.2em}
\begin{columns}[T, totalwidth=\textwidth]

  \begin{column}{0.47\textwidth}
    \medskip
    \stepcircle{1}\enspace{\bfseries\color{cteal} Normalise field}\\[2pt]
    \hspace{1.6em}{\small z-score normalise to zero-mean unit-variance}

    \bigskip
    \stepcircle{2}\enspace{\bfseries\color{cteal} Fit hyperparameters}\\[2pt]
    \hspace{1.6em}{\small Marginal likelihood on 1{,}500-pt subsample
      $\;\to\;$ \texttt{LS\_Z}, \texttt{LS\_XY}}

    \bigskip
    \stepcircle{3}\enspace{\bfseries\color{cteal} Select anchors}\\[2pt]
    \hspace{1.6em}{\small rpgks: randomised pivoted Cholesky
      $+$ QR pivot on low-rank factor $F$}
  \end{column}

  \begin{column}{0.06\textwidth}
    \color{cteal}\rule{1pt}{0.75\textheight}
  \end{column}

  \begin{column}{0.47\textwidth}
    \medskip
    \stepcircle{4}\enspace{\bfseries\color{cteal} GP prediction}\\[2pt]
    \hspace{1.6em}{\small Posterior mean $+$ variance at all candidate
      points (batched)}

    \bigskip
    \stepcircle{5}\enspace{\bfseries\color{cteal} Accept within $\pm\varepsilon_b$}\\[2pt]
    \hspace{1.6em}{\small Points with $|\text{err}| < \varepsilon_b$
      marked compressed; pool shrinks}

    \bigskip
    \stepcircle{6}\enspace{\bfseries\color{cteal} Repeat}\\[2pt]
    \hspace{1.6em}{\small Next round targets residual high-variance
      region with 200 new anchors}
  \end{column}

\end{columns}
\end{frame}
