% ── SLIDE 2: Problem + Prior Solution + SZ2 Weaknesses ───────
\begin{frame}{The Problem — and One Prior Solution}
\begin{columns}[T]

  % Block 1: Problem
  \begin{column}{0.31\textwidth}
    \begingroup
    \setbeamercolor{block title}{bg=cred!15, fg=cred}
    \setbeamercolor{block body}{bg=cred!5, fg=cnavy}
    \begin{block}{The Problem}
      \footnotesize
      Scientific simulations produce \textbf{more data than we can store}.
      A single climate or physics run can generate terabytes per hour —
      far beyond disk and bandwidth budgets.
    \end{block}
    \endgroup
  \end{column}

  % Block 2: Prior solution
  \begin{column}{0.34\textwidth}
    \begingroup
    \setbeamercolor{block title}{bg=cice, fg=cnavy}
    \setbeamercolor{block body}{bg=cice!40, fg=cnavy}
    \begin{block}{One Solution: Prediction-Based Compression}
      \footnotesize
      \textbf{Predict} $\to$ \textbf{residual} $\to$ \textbf{quantize}
      $\to$ \textbf{entropy code}\\[0.3em]
      Example: SZ2 (Liang et al., 2018), ZFP (Lindstrom, 2014)\\[0.2em]
    \end{block}
    \endgroup
  \end{column}

  % Block 3: SZ2 weaknesses
  \begin{column}{0.31\textwidth}
    \begingroup
    \setbeamercolor{block title}{bg=camber!25, fg=cnavy}
    \setbeamercolor{block body}{bg=camber!10, fg=cnavy}
    \begin{block}{What SZ2 Misses}
      \footnotesize
      \begin{itemize}\setlength\itemsep{0.2em}
        \item \textbf{Local only} — predicts from 2–3 neighbors; no global pattern learning
        \item \textbf{No training} — cannot learn structure across fields or time
        \item \textbf{One snapshot at a time} — no cross-field or temporal correlation
      \end{itemize}
    \end{block}
    \endgroup
  \end{column}

\end{columns}

\vspace{0.4em}
{\footnotesize\color{cmuted}
  \textbf{Our question:} Can we do better by learning global spatial structure
  from a small number of sensor observations?
}
\end{frame}

% ── SLIDE 3: Our Approach ─────────────────────────────────────
\begin{frame}{Our Approach — Sensor-Based Reconstruction}
\begin{columns}[T]

  \begin{column}{0.42\textwidth}
    {\small\textcolor{cteal}{\textbf{Key idea:}} choose $k$ sensor locations,
    observe the field there, and reconstruct everywhere.}\\[0.25em]

    \begingroup
    \setbeamercolor{block title}{bg=cgreen!15, fg=cgreen}
    \setbeamercolor{block body}{bg=cgreen!5, fg=cnavy}
    \begin{block}{Two families of methods}
      \footnotesize
      \textbf{DEIM} — learn dominant spatial modes (SVD) from training snapshots;
      sensors selected by Q-DEIM pivoting.\\[0.15em]
      \textbf{Kriging} — fit a Matérn Gaussian Process; sensors selected
      by Randomly Pivoted Cholesky.
    \end{block}
    \endgroup

    \vspace{0.2em}
    {\small\textbf{What we store:}}\\
    {\footnotesize basis/model (float16) + sensor indices + sensor values (float16)
    + quantized residuals}
  \end{column}

  \begin{column}{0.56\textwidth}

% ── SLIDE 4: Results ──────────────────────────────────────────
\begin{frame}{Results — PSNR vs Compression Ratio}
\begin{columns}[T]

  \begin{column}{0.56\textwidth}
    \vspace{-0.5em}
    \begin{center}
      \includegraphics[width=\textwidth, height=0.78\textheight, keepaspectratio]%
        {poster_rd_psnr_CLOUDf48.png}
    \end{center}
  \end{column}

  \begin{column}{0.42\textwidth}
    \vspace{0.4em}
    {\small\textcolor{cteal}{\textbf{Reading the plot}}}\\[0.25em]
    {\footnotesize
    \textbf{PSNR} (y-axis) — reconstruction accuracy in decibels.
    Higher is better; each +6 dB $\approx$ halves the RMS error.\\[0.25em]
    \textbf{CR} (x-axis, log) = raw bytes / stored bytes.
    Higher means more compressed.
    }

    \vspace{0.35em}
    \begingroup
    \setbeamercolor{block title}{bg=camber!25, fg=cnavy}
    \setbeamercolor{block body}{bg=camber!10, fg=cnavy}
    \begin{block}{Takeaway}
      \footnotesize
      Our methods achieve \textbf{higher accuracy} at the compression
      ratios where they operate. SZ2 and ZFP reach higher CR because
      their local predictor produces low-entropy residuals — closing
      this gap is our primary future work.
    \end{block}
    \endgroup
  \end{column}

\end{columns}
\end{frame}
